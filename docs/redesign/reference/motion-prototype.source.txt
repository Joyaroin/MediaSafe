<script type="text/x-dc" data-dc-script data-props='{"$preview":{"width":800,"height":844}}'>
class Component extends DCLogic {
fresh() {
return {
hero: { phase: 'rest', idx: 0, done: false },
rows: {
sev: { phase: 'rest', ep: 0 },
and: { phase: 'rest', ep: 0, retried: false },
sho: { phase: 'rest', ep: 0 },
slo: { phase: 'rest', ep: 0 },
past: { phase: 'rest', ep: 0 }
},
toast: false,
tv: null,
nodComing: 0,
nodLibrary: 0
};
}
now() {
return this.state && this.state.hero ? this.state : this.fresh();
}
later(key, ms, fn) {
if (!this.timers) this.timers = {};
if (this.timers[key]) clearTimeout(this.timers[key]);
this.timers[key] = setTimeout(fn, ms);
}
cancel(key) {
if (this.timers && this.timers[key]) { clearTimeout(this.timers[key]); this.timers[key] = null; }
}
componentWillUnmount() {
if (!this.timers) return;
for (var key in this.timers) { if (this.timers[key]) clearTimeout(this.timers[key]); }
}
setRow(id, patch) {
var s = this.now();
var rows = Object.assign({}, s.rows);
rows[id] = Object.assign({}, rows[id], patch);
this.setState(Object.assign({}, s, { rows: rows }));
}
renderVals() {
var self = this;
var s = this.now();
var T = '#f3f0ea', D = '#bab5ac', G = '#181715';
var SERIF = "'Instrument Serif', Georgia, serif";
var HERO = ['S1 E2 · Hands', 'S1 E3 · Brigade', 'S1 E4 · Dogs', 'S1 E5 · Sheridan', 'S1 E6 · Ceres', 'S1 E7 · Review', 'S1 E8 · Braciole'];
var ORDER = ['sev', 'and', 'sho', 'slo', 'past'];
var DATA = {
sev: { kind: 'next', bg: '#1f6b70', eps: ['S1 E2 · Half Loop', 'S1 E3 · In Perpetuity', 'S1 E4 · The You You Are'] },
and: { kind: 'fail', bg: '#8a4a2a', eps: ['S1 E2 · That Would Be Me', 'S1 E3 · Reckoning', 'S1 E4 · Aldhani'] },
sho: { kind: 'finish', bg: '#8c2f3f', eps: ['S1 E10 · A Dream of a Dream'] },
slo: { kind: 'next', bg: '#5a4a78', eps: ['S1 E2 · Work Drinks', 'S1 E3 · Bad Tradecraft', 'S1 E4 · Visiting Hours'] },
past: { kind: 'film', bg: '#6a3f7a', eps: ['Film · 2023'] }
};

function leave(id, dest) {
var cur = self.now();
var before = 0;
for (var i = 0; i < ORDER.length; i++) {
if (ORDER[i] === id) break;
if (cur.rows[ORDER[i]].phase !== 'gone') before++;
}
var fromY = 408 + 64 * before + 5;
var toX = dest === 'coming' ? 80.6 : 233.4;
var rows = Object.assign({}, cur.rows);
rows[id] = Object.assign({}, rows[id], { phase: 'gone' });
self.setState(Object.assign({}, cur, { rows: rows, tv: { bg: DATA[id].bg, fromY: fromY, toX: toX, go: false } }));
self.later('tv-go', 40, function () {
var c = self.now();
if (!c.tv) return;
self.setState(Object.assign({}, c, { tv: Object.assign({}, c.tv, { go: true }) }));
});
self.later('tv-nod', 360, function () {
var c = self.now();
var patch = dest === 'coming' ? { nodComing: c.nodComing + 1 } : { nodLibrary: c.nodLibrary + 1 };
self.setState(Object.assign({}, c, patch));
});
self.later('tv-end', 520, function () {
var c = self.now();
self.setState(Object.assign({}, c, { tv: null }));
});
}
function commitRow(id) {
var cur = self.now();
var r = cur.rows[id];
if (r.phase !== 'held') return;
var kind = DATA[id].kind;
if (kind === 'finish') {
self.setRow(id, { phase: 'finished' });
self.later(id + '-leave', 1100, function () { leave(id, 'coming'); });
return;
}
if (kind === 'film') { leave(id, 'library'); return; }
var next = Math.min(r.ep + 1, DATA[id].eps.length - 1);
self.setRow(id, { phase: 'rest', ep: next });
}
function markRow(id) {
var cur = self.now();
var r = cur.rows[id];
if (r.phase !== 'rest' && r.phase !== 'failed') return;
self.setRow(id, { phase: 'held' });
if (DATA[id].kind === 'fail' && !r.retried) {
self.later(id, 700, function () {
var c = self.now();
if (c.rows[id].phase !== 'held') return;
var rows = Object.assign({}, c.rows);
rows[id] = Object.assign({}, rows[id], { phase: 'failed' });
self.setState(Object.assign({}, c, { rows: rows, toast: true }));
});
return;
}
self.later(id, 1200, function () { commitRow(id); });
}
function undoRow(id) {
var cur = self.now();
if (cur.rows[id].phase !== 'held') return;
self.cancel(id);
self.setRow(id, { phase: 'rest' });
}
function rowView(id) {
var r = s.rows[id];
var gone = r.phase === 'gone';
var fin = r.phase === 'finished';
var filled = r.phase === 'held' || fin;
return {
maxH: gone ? '0px' : '64px',
op: gone ? 0 : 1,
line: fin ? 'Season 1, finished' : DATA[id].eps[r.ep],
lineAnim: (r.ep + (fin ? 1 : 0)) % 2 === 0 ? 'ms-roll-a' : 'ms-roll-b',
lineFont: fin ? SERIF : 'inherit',
lineStyle: fin ? 'italic' : 'normal',
lineSize: fin ? '18px' : '13px',
lineColor: fin ? T : D,
undoOp: r.phase === 'held' ? 1 : 0,
undoPe: r.phase === 'held' ? 'auto' : 'none',
tickBg: filled ? T : 'rgba(243,240,234,0)',
tickBorder: filled ? T : D,
tickColor: filled ? G : D,
tickDur: r.phase === 'failed' ? '420ms' : '160ms',
popAnim: filled ? 'ms-pop' : 'none',
ringOp: r.phase === 'held' ? 1 : 0,
drain: r.phase === 'held' ? 'ms-drain-row' : 'none',
mark: function () { markRow(id); },
undo: function () { undoRow(id); }
};
}

var h = s.hero;
var held = h.phase === 'held';
var watched = h.done ? 8 : 1 + h.idx + (held ? 1 : 0);
var seg = { gap: h.done ? '0px' : '4px' };
for (var n = 1; n <= 8; n++) seg['c' + n] = n <= watched ? T : 'rgba(255,255,255,0.22)';

var tv = s.tv;
var tvView = tv
? {
bg: tv.bg,
o: tv.go ? 0 : 1,
t: tv.go ? 'translate(' + tv.toX + 'px, 764.5px) scale(0.22)' : 'translate(0px, ' + tv.fromY + 'px) scale(1)',
tr: tv.go ? 'transform 420ms cubic-bezier(0.65,0,0.35,1), opacity 420ms ease-in' : 'none'
}
: { bg: 'rgba(0,0,0,0)', o: 0, t: 'translate(0px, 0px) scale(1)', tr: 'none' };

return {
hero: {
w: held ? '56px' : '350px',
r: held ? '28px' : '14px',
gap: held ? '0px' : '10px',
labelW: held ? '0px' : '170px',
labelOp: held ? 0 : 1,
btnOp: h.done ? 0 : 1,
btnPe: held || h.done ? 'none' : 'auto',
ringOp: held ? 1 : 0,
drain: held ? 'ms-drain-hero' : 'none',
infoOp: held ? 1 : 0,
infoPe: held ? 'auto' : 'none',
line: h.done ? 'Season 1, finished' : HERO[h.idx],
lineAnim: (h.idx + (h.done ? 1 : 0)) % 2 === 0 ? 'ms-roll-a' : 'ms-roll-b',
lineFont: h.done ? SERIF : 'inherit',
lineStyle: h.done ? 'italic' : 'normal',
lineSize: h.done ? '24px' : '17px',
count: watched + ' of 8',
countAnim: watched % 2 === 0 ? 'ms-roll-a' : 'ms-roll-b'
},
seg: seg,
heroMark: function () {
var c = self.now();
if (c.hero.phase !== 'rest' || c.hero.done) return;
self.setState(Object.assign({}, c, { hero: Object.assign({}, c.hero, { phase: 'held' }) }));
self.later('hero', 1200, function () {
var d = self.now();
if (d.hero.phase !== 'held') return;
var last = d.hero.idx >= HERO.length - 1;
self.setState(Object.assign({}, d, { hero: { phase: 'rest', idx: last ? d.hero.idx : d.hero.idx + 1, done: last } }));
});
},
heroUndo: function () {
var c = self.now();
if (c.hero.phase !== 'held') return;
self.cancel('hero');
self.setState(Object.assign({}, c, { hero: Object.assign({}, c.hero, { phase: 'rest' }) }));
},
r0: rowView('sev'),
r1: rowView('and'),
r2: rowView('sho'),
r3: rowView('slo'),
r4: rowView('past'),
tv: tvView,
toast: {
o: s.toast ? 1 : 0,
t: s.toast ? 'translateY(0px) scale(1)' : 'translateY(-120px) scale(0.92)',
pe: s.toast ? 'auto' : 'none'
},
toastRetry: function () {
var c = self.now();
var rows = Object.assign({}, c.rows);
rows.and = Object.assign({}, rows.and, { retried: true });
self.setState(Object.assign({}, c, { rows: rows, toast: false }));
self.later('retry', 60, function () { markRow('and'); });
},
toastDismiss: function () {
var c = self.now();
self.setState(Object.assign({}, c, { toast: false }));
},
nod: {
coming: s.nodComing === 0 ? 'none' : s.nodComing % 2 === 1 ? 'ms-nod-a' : 'ms-nod-b',
library: s.nodLibrary === 0 ? 'none' : s.nodLibrary % 2 === 1 ? 'ms-nod-a' : 'ms-nod-b',
comingColor: D,
libraryColor: D
},
reset: function () {
if (self.timers) { for (var key in self.timers) { if (self.timers[key]) clearTimeout(self.timers[key]); } }
self.timers = {};
self.setState(self.fresh());
}
};
}
}
